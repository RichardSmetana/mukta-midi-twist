# sonic-pi-01-led-test.rb
# =============================================================================
# Mukta MIDI Twist — LED test for Sonic Pi
# =============================================================================
#
# Sends MIDI notes to onboard RGB (GPIO 48) and WS2812 strip (GPIO 7).
# Note map: see ../config.h (onboard 45–47, strip 48–59).
#
# Prerequisites:
#   - Firmware running, "Mukta MIDI Twist" enabled in Sonic Pi → Preferences → I/O
#   - MIDI port name below must match your system (may differ from example)
#
# Behaviour (unchanged):
#   - 60 BPM, live_loop :test1
#   - :index toggles 0/1 → base note 45 or 46
#   - 4× per cycle: cymbal + MIDI flash per strip pixel, 0.25 s on/off
# =============================================================================

use_bpm 60

set :flag, false
set :index, 0

live_loop :test1 do
  index = get(:index)
  note1 = 45 + index

  4.times do |i|
    sample :drum_cymbal_closed, amp: (i == 0 ? 1.0 : 0.5)

    5.times do |j|
      midi_note_on note1 + j * 3, vel_f: 0.2, port: "mukta_midi_twist__0_"
    end
    midi_note_on 48 + i * 3, vel_f: 0.6, port: "mukta_midi_twist__0_"
    midi_note_on 49 + i * 3, vel_f: 0.6, port: "mukta_midi_twist__0_"
    midi_note_on 50 + i * 3, vel_f: 0.6, port: "mukta_midi_twist__0_"
    # midi_note_on 48 + i * 3 + index, vel_f: 0.2, port: "mukta_midi_twist__0_"

    sleep 0.25

    5.times do |j|
      midi_note_off note1 + j * 3, port: "mukta_midi_twist__0_"
    end
    midi_note_off 48 + i * 3, port: "mukta_midi_twist__0_"
    midi_note_off 49 + i * 3, port: "mukta_midi_twist__0_"
    midi_note_off 50 + i * 3, port: "mukta_midi_twist__0_"
    # midi_note_off 48 + i * 3 + index, port: "mukta_midi_twist__0_"

    sleep 0.25
  end

  set :index, get(:index) + 1
  set :index, 0 if get(:index) >= 2
end
